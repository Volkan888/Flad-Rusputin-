// China-Telefon Easter Egg (nach NATOHotline einfügen):

static class ChinaHotline
{
    static Random rand = new Random();
    
    public static void CallChina(PlayerCharacter p)
    {
        if (!p.ChinaTelefonAktiv)
        {
            Console.Clear();
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine("\n📞 CHINA-TELEFON");
            Console.WriteLine("═══════════════════════════════════════════════════════\n");
            Console.ResetColor();
            Console.WriteLine("Das China-Telefon ist noch nicht freigeschaltet!");
            Console.WriteLine("\n💡 Tipp: Wird durch China-Events aktiviert.");
            Console.WriteLine("\n[Drücke eine Taste...]");
            Console.ReadKey(true);
            return;
        }
        
        if (p.ChinaAnrufeVerfügbar <= 0)
        {
            Console.Clear();
            Console.ForegroundColor = ConsoleColor.Red;
            Console.WriteLine("\n📞 CHINA-TELEFON");
            Console.WriteLine("═══════════════════════════════════════════════════════\n");
            Console.ResetColor();
            Console.WriteLine("Alle Anrufe aufgebraucht!");
            Console.WriteLine("\n[Drücke eine Taste...]");
            Console.ReadKey(true);
            return;
        }
        
        Console.Clear();
        Console.ForegroundColor = ConsoleColor.Red;
        Console.WriteLine(@"
        ╔═══════════════════════════════════╗
        ║   📞 CHINA-TELEFON 📞            ║
        ║   🐉 DRACHEN-HOTLINE 🐉          ║
        ╚═══════════════════════════════════╝
        ");
        Console.ResetColor();
        
        Console.WriteLine($"🔋 Verbleibende Anrufe: {p.ChinaAnrufeVerfügbar}/3");
        Console.WriteLine($"🤝 China-Beziehung: {p.ChinaBeziehung}%");
        Console.WriteLine();
        
        Console.WriteLine("[1] 🐉 'Hallo, guter Freund!' (Diplomatisch)");
        Console.WriteLine("[2] 🤷 'Was soll der Anruf?' (Überrascht)");
        Console.WriteLine("[3] 🐼 'Panda-Freunde!' (Krawallstour)");
        Console.WriteLine("[4] 🔙 Zurück");
        Console.Write("\nWähle [1-4]: ");
        
        string choice = Console.ReadLine();
        
        switch (choice)
        {
            case "1":
                GuterFreund(p);
                break;
            case "2":
                Ueberrascht(p);
                break;
            case "3":
                Krawallstour(p);
                break;
            case "4":
                return;
        }
        
        p.ChinaAnrufeVerfügbar--;
    }
    
    static void GuterFreund(PlayerCharacter p)
    {
        Console.Clear();
        Console.ForegroundColor = ConsoleColor.Green;
        Console.WriteLine("\n🐉 DIPLOMATISCHER ANSATZ");
        Console.WriteLine("═══════════════════════════════════════════════════════\n");
        Console.ResetColor();
        
        Console.WriteLine("📞 *Ring Ring*");
        Thread.Sleep(1000);
        Console.ForegroundColor = ConsoleColor.Yellow;
        Console.WriteLine("\nXi Jinping: '同志！(Genosse!) Herzliche Grüße aus Peking!'");
        Console.WriteLine("Xi: 'Gratuliere zum Drachenboot-Rennen... äh, zu Ihren Erfolgen!'");
        Console.WriteLine("Xi: 'Wir möchten über Sibirien-Rohstoffe sprechen...'");
        Console.ResetColor();
        
        Console.WriteLine("\n[1] Handelsabkommen annehmen");
        Console.WriteLine("[2] Höflich vertagen");
        Console.Write("\nWähle [1-2]: ");
        
        if (Console.ReadLine() == "1")
        {
            Console.Clear();
            Console.ForegroundColor = ConsoleColor.Green;
            Console.WriteLine("\n✓ HANDELSABKOMMEN!");
            Console.WriteLine("═══════════════════════════════════════════════════════\n");
            
            p.Geld += 400;
            p.ChinaBeziehung += 20;
            
            Console.WriteLine("Xi: '非常好! (Sehr gut!) Russland-China Freundschaft!'");
            Console.WriteLine("\n💰 Geld: +400 Rubel (Rohstoffhandel)");
            Console.WriteLine($"🤝 China-Beziehung: +20% (jetzt {p.ChinaBeziehung}%)");
            Console.WriteLine("\n✨ Wirtschaftszusammenarbeit freigeschaltet!");
            Console.ResetColor();
        }
        else
        {
            Console.WriteLine("\nXi: 'Verstanden. Die Tür bleibt offen.'");
            p.ChinaBeziehung += 5;
        }
        
        Console.WriteLine("\n[Drücke eine Taste...]");
        Console.ReadKey(true);
    }
    
    static void Ueberrascht(PlayerCharacter p)
    {
        Console.Clear();
        Console.ForegroundColor = ConsoleColor.Yellow;
        Console.WriteLine("\n🤷 ÜBERRASCHTER ANSATZ");
        Console.WriteLine("═══════════════════════════════════════════════════════\n");
        Console.ResetColor();
        
        Console.WriteLine("📞 *Ring Ring*");
        Thread.Sleep(1000);
        Console.ForegroundColor = ConsoleColor.Yellow;
        Console.WriteLine("\nXi: 'Genosse, wir haben 10.000 Tonnen Reis für Sie!'");
        Console.WriteLine("Xi: 'Als Zeichen unserer Freundschaft...'");
        Console.ResetColor();
        
        Console.WriteLine("\n[1] Reis annehmen");
        Console.WriteLine("[2] Höflich ablehnen");
        Console.Write("\nWähle [1-2]: ");
        
        if (Console.ReadLine() == "1")
        {
            Console.WriteLine("\n✓ 10.000 Tonnen Reis geliefert!");
            p.Geld += 100;
            p.ChinaBeziehung += 10;
            Console.WriteLine("\n💰 Geld: +100 Rubel (Reiswert)");
        }
        else
        {
            Console.WriteLine("\nXi: 'Schade... aber kein Problem!'");
            Console.WriteLine("\n→ Verpasste günstige Import-Chance");
        }
        
        Console.WriteLine("\n[Drücke eine Taste...]");
        Console.ReadKey(true);
    }
    
    static void Krawallstour(PlayerCharacter p)
    {
        Console.Clear();
        Console.ForegroundColor = ConsoleColor.Red;
        Console.WriteLine("\n🐼 KRAWALLSTOUR!");
        Console.WriteLine("═══════════════════════════════════════════════════════\n");
        Console.ResetColor();
        
        Console.WriteLine("📞 Du wählst unhöflichen Ton...");
        Thread.Sleep(1000);
        Console.WriteLine("\n'Nun hör mal, Ihr Panda-Freunde!'");
        Console.WriteLine("'Ni hao, ma? Eier von Pekingente, sonst keine Finanzhilfe!'");
        Thread.Sleep(1500);
        
        Console.ForegroundColor = ConsoleColor.Red;
        Console.WriteLine("\n💥 DIPLOMATISCHER EKLAT!");
        Console.WriteLine("═══════════════════════════════════════════════════════\n");
        
        Console.WriteLine("Xi: '...' *legt auf*");
        Console.WriteLine("\n📰 Chinesische Medien berichten über 'Beleidigung'!");
        
        p.ChinaBeziehung -= 40;
        p.Geld -= 300; // Strafzölle
        p.EinflussInternational -= 20;
        
        Console.WriteLine($"\n🤝 China-Beziehung: -40% (jetzt {p.ChinaBeziehung}%)");
        Console.WriteLine($"💰 Geld: -300 Rubel (Handelssperre!)");
        Console.WriteLine($"🌍 Internationales Ansehen: -20");
        Console.WriteLine("\n⚠️  Handelsembargo für 5 Jahre!");
        Console.ResetColor();
        
        Console.WriteLine("\n[Drücke eine Taste...]");
        Console.ReadKey(true);
    }
}
